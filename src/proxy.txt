import { NextResponse, NextRequest } from "next/server";

const BACKEND_URL =
  process.env.NEXT_PUBLIC_BACKEND_URL || "http://localhost:4000";

export async function proxy(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Ignorar archivos internos de Next y la API
  if (
    pathname.startsWith("/_next") ||
    pathname.startsWith("/api") ||
    pathname === "/favicon.ico"
  ) {
    return NextResponse.next();
  }

  // Rutas públicas
  const publicRoutes = ["/", "/login", "/register"];
  const isPublic = publicRoutes.includes(pathname);

  // Construir header de cookies correctamente
  const cookieHeader = request.cookies
    .getAll()
    .map(({ name, value }) => `${name}=${value}`)
    .join("; ");

  // Obtener usuario desde el backend
  let user = null;
  try {
    const res = await fetch(`${BACKEND_URL}/api/users/me`, {
      method: "GET",
      headers: { cookie: cookieHeader },
    });

    if (res.ok) {
      user = await res.json();
    }
  } catch (err) {
    console.error("Error verificando sesión:", err);
  }

  const isLoggedIn = !!user;

  // 1️⃣ Ruta pública + logueado → redirigir al dashboard correspondiente
  if (isPublic && isLoggedIn) {
    // Redirigir según rol
    if (user.role === "ADMIN")
      return NextResponse.redirect(new URL("/assignTicket", request.url));
    if (user.role === "AGENT")
      return NextResponse.redirect(new URL("/agent", request.url));
    return NextResponse.redirect(new URL("/dashboard", request.url));
  }

  // 2️⃣ Ruta privada + NO logueado → login
  if (!isPublic && !isLoggedIn) {
    return NextResponse.redirect(new URL("/login", request.url));
  }

  // 3️⃣ Protección por roles
  if (pathname.startsWith("/assignTicket") && user?.role !== "ADMIN") {
    return NextResponse.redirect(new URL("/dashboard", request.url));
  }

  if (pathname.startsWith("/agent") && user?.role !== "AGENT") {
    return NextResponse.redirect(new URL("/dashboard", request.url));
  }

  // Opcional: otras protecciones según rutas futuras

  return NextResponse.next();
}

export const config = {
  matcher: ["/((?!_next|favicon.ico|api/).*)"],
};
